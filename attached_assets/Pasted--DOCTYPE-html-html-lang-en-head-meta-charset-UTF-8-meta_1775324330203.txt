  <script>
    async function loadRoleplays() {
      const res = await fetch('/api/roleplays?category=difficult-guests');
      const data = await res.json();
      const container = document.getElementById('roleplays-container');
      container.innerHTML = data.map(rp => `
        <div class="bg-white rounded-3xl p-8 shadow-sm hover:shadow transition">
          <h3 class="font-semibold text-xl mb-4">${rp.title}</h3>
          <p class="text-gray-600 mb-6 line-clamp-3">${rp.setup}</p>
          <button onclick="openRoleplay(${rp.id})" 
                  class="w-full py-4 bg-blue-600 text-white rounded-2xl hover:bg-blue-700">
            Start Role-Play
          </button>
        </div>
      `).join('');
    }

    async function loadQuizzes() {
      const res = await fetch('/api/quizzes?module=wine-service');
      const data = await res.json();
      const container = document.getElementById('quizzes-container');
      container.innerHTML = data.map(q => `
        <div class="bg-white rounded-3xl p-8 shadow-sm">
          <h3 class="font-semibold text-xl mb-4">${q.title}</h3>
          <button onclick="startQuiz(${q.id})" 
                  class="mt-4 px-8 py-3 bg-green-600 text-white rounded-xl hover:bg-green-700">
            Take Quiz
          </button>
        </div>
      `).join('');
    }

    let currentRoleplay = null;

    async function openRoleplay(id) {
      const res = await fetch('/api/roleplays?category=difficult-guests');
      const data = await res.json();
      currentRoleplay = data.find(r => r.id === id);
      if (!currentRoleplay) return;

      document.getElementById('modal-title').textContent = currentRoleplay.title;
      document.getElementById('modal-content').innerHTML = `
        <p class="mb-6"><strong>Setup:</strong> ${currentRoleplay.setup}</p>
        <div class="bg-gray-50 p-6 rounded-2xl mb-8 whitespace-pre-line">${currentRoleplay.dialogue}</div>
        <p><strong>Debrief:</strong> ${currentRoleplay.debrief}</p>
      `;
      document.getElementById('roleplay-modal').classList.add('show');
    }

    function closeModal() {
      document.getElementById('roleplay-modal').classList.remove('show');
    }

    function markAsComplete() {
      alert("Progress saved! (Connect to your progress API here)");
      closeModal();
      loadRoleplays();
    }

    function startQuiz(id) {
      alert(`Starting quiz ${id} — expand this later with full quiz UI`);
    }

    // Load content when page opens
    window.onload = () => {
      loadRoleplays();
      loadQuizzes();
    };
  </script>
