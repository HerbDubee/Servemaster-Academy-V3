<!-- Stripe + Dynamic Currency Logic -->
<script src="https://js.stripe.com/v3/"></script>
<script>
  const stripe = Stripe('pk_live_XXXXXXXXXXXXXXXXXXXXXXXX'); // ← PASTE YOUR LIVE PUBLISHABLE KEY HERE

  const elements = stripe.elements({
    mode: 'subscription',
    amount: 1900,
    currency: 'cad',
    appearance: { theme: 'stripe' }
  });

  const currencySelector = elements.create('currencySelector', {
    defaultCurrency: 'cad',
    supportedCurrencies: ['cad', 'usd']
  });
  currencySelector.mount('#currency-selector');

  // Dynamic price update
  currencySelector.on('change', (event) => {
    const newCurrency = event.currency.toUpperCase();
    const rate = event.exchangeRate || 1;

    document.querySelectorAll('.price-display').forEach(el => {
      const base = parseFloat(el.dataset.baseAmount);
      const converted = (base * rate).toFixed(2);
      el.innerHTML = `${converted} <span class="text-xl text-zinc-400">${newCurrency} / ${el.closest('div').querySelector('h3').textContent.includes('Monthly') ? 'mo' : 'yr'}</span>`;
    });
  });

  async function checkout(priceId) {
    const res = await fetch('/create-checkout-session', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ priceId })
    });
    const { sessionId } = await res.json();
    stripe.redirectToCheckout({ sessionId });
  }
</script>
