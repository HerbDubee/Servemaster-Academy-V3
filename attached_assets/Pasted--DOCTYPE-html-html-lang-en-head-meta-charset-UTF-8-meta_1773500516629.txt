        function calculateROI() {
          const servers = parseFloat(document.getElementById('servers').value) || 0;
          const cost = parseFloat(document.getElementById('cost').value) || 0;
          const savings = Math.round(servers * cost * 0.75); // 75% faster onboarding assumption
          document.getElementById('result').innerHTML = `You could save <span class="text-green-600">$${savings.toLocaleString()}</span> in Year 1`;
