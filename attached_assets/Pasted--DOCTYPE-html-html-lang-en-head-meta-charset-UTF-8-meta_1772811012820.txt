    // Show trial days left if logged in
    async function showTrialBanner() {
      const token = localStorage.getItem('token');
      if (!token) return;
      
      const res = await fetch('/login', { 
        headers: { Authorization: `Bearer ${token}` } 
      });
      const data = await res.json();
      
      if (data.trialDaysLeft > 0) {
        document.getElementById('trial-banner').classList.remove('hidden');
        document.getElementById('days-left').textContent = data.trialDaysLeft;
      }
    }
    showTrialBanner();
