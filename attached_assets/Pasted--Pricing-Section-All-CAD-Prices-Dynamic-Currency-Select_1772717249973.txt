<!-- Stripe.js + Elements + Currency Selector Logic -->
<script src="https://js.stripe.com/v3/"></script>
<script>
  const stripe = Stripe('pk_live_xxxxxxxxxxxxxxxxxxxxxxxx'); // ← REPLACE WITH YOUR LIVE PUBLISHABLE KEY

  // Create Elements instance
  const elements = stripe.elements({
    mode: 'subscription',
    amount: 1900, // $19 CAD in cents – Stripe adjusts dynamically
    currency: 'cad', // Default to CAD
    appearance: {
      theme: 'stripe',
      variables: {
        colorPrimary: '#d4af37', // your gold accent
        colorBackground: '#111827',
        colorText: '#ffffff'
      }
    }
  });

  // Mount Currency Selector Element
  const currencySelector = elements.create('currencySelector', {
    defaultCurrency: 'cad',
    supportedCurrencies: ['cad', 'usd'] // Allow CAD and USD switching
  });
  currencySelector.mount('#currency-selector');

  // Dynamic price formatting when currency changes
  currencySelector.on('change', (event) => {
    const newCurrency = event.currency.toUpperCase();
    const exchangeRate = event.exchangeRate || 1; // Stripe provides approximate rate

    document.querySelectorAll('.price-display').forEach(el => {
      const baseAmount = parseFloat(el.dataset.baseAmount);
      const converted = (baseAmount * exchangeRate).toFixed(2);
      const symbol = newCurrency;

      el.innerHTML = `${converted} <span class="text-xl text-zinc-400 currency-symbol">${symbol} / ${el.closest('div').querySelector('h3').textContent.includes('Monthly') ? 'mo' : 'yr'}</span>`;
    });
  });

  // Checkout function
  async function checkout(priceId) {
    try {
      const response = await fetch('/create-checkout-session', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ priceId })
      });
      const { sessionId } = await response.json();
      const { error } = await stripe.redirectToCheckout({ sessionId });
      if (error) console.error(error.message);
    } catch (err) {
      console.error('Checkout error:', err);
    }
  }
</script>
