  <div id="roleplay-modal" class="modal">
    <div class="bg-white rounded-3xl max-w-2xl w-full mx-4 p-8 max-h-[90vh] overflow-y-auto">
      <div class="flex justify-between mb-6">
        <h3 id="modal-title" class="text-2xl font-bold"></h3>
        <button onclick="closeModal()" class="text-4xl text-gray-400 hover:text-gray-600">×</button>
      </div>

      <div id="modal-content" class="prose text-gray-700 mb-10"></div>

      <!-- TTS Controls -->
      <div class="bg-gray-50 p-6 rounded-2xl mb-8">
        <p class="text-sm text-gray-500 mb-4 font-medium">Listen to the dialogue</p>
        <div class="grid grid-cols-3 gap-3">
          <button onclick="speakServerLines()" class="py-3 bg-blue-600 text-white rounded-xl hover:bg-blue-700">Server Lines</button>
          <button onclick="speakGuestLines()" class="py-3 bg-amber-600 text-white rounded-xl hover:bg-amber-700">Guest Lines</button>
          <button onclick="speakFullDialogue()" class="py-3 bg-purple-600 text-white rounded-xl hover:bg-purple-700">Full Dialogue</button>
        </div>
      </div>

      <div class="flex gap-4">
        <button onclick="closeModal()" class="flex-1 py-4 bg-gray-200 hover:bg-gray-300 rounded-2xl font-medium">Close</button>
        <button onclick="markAsComplete()" class="flex-1 py-4 bg-green-600 hover:bg-green-700 text-white rounded-2xl font-medium">Mark Completed</button>
      </div>
    </div>

  <script>
    let currentRoleplay = null;

    async function loadRoleplays() {
      try {
        const res = await fetch('/api/roleplays?category=difficult-guests');
        const data = await res.json();
        const container = document.getElementById('roleplays-container');
        container.innerHTML = data.map(rp => `
          <div class="bg-white rounded-3xl p-8 shadow-sm hover:shadow-md transition card" onclick="openRoleplay(${rp.id})">
            <h3 class="font-semibold text-xl mb-4">${rp.title}</h3>
            <p class="text-gray-600 mb-6 line-clamp-3">${rp.setup}</p>
            <div class="text-blue-600 text-sm font-medium">Click to practice with voice →</div>
          </div>
        `).join('');
      } catch (e) {
        console.error(e);
      }
    }

    async function openRoleplay(id) {
      const res = await fetch('/api/roleplays?category=difficult-guests');
      const data = await res.json();
      currentRoleplay = data.find(r => r.id === id);
      if (!currentRoleplay) return;

      document.getElementById('modal-title').textContent = currentRoleplay.title;
      document.getElementById('modal-content').innerHTML = `
        <p class="mb-6"><strong>Setup:</strong> ${currentRoleplay.setup}</p>
        <div class="bg-gray-50 p-6 rounded-2xl mb-8 whitespace-pre-line text-[15px] leading-relaxed">${currentRoleplay.dialogue}</div>
        <p class="text-gray-700"><strong>Debrief:</strong> ${currentRoleplay.debrief}</p>
      `;
      document.getElementById('roleplay-modal').classList.add('show');
    }

    function closeModal() {
      document.getElementById('roleplay-modal').classList.remove('show');
    }

    // TTS Functions
    function speak(text) {
      if ('speechSynthesis' in window) {
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.rate = 0.95;
        utterance.pitch = 1.05;
        speechSynthesis.speak(utterance);
      } else {
        alert("Text-to-Speech is not supported in this browser.");
      }
    }

    function speakServerLines() {
      if (!currentRoleplay) return;
      const serverText = currentRoleplay.dialogue.split('\n')
        .filter(line => line.includes('Server:'))
        .join('. ');
      speak(serverText);
    }

    function speakGuestLines() {
      if (!currentRoleplay) return;
      const guestText = currentRoleplay.dialogue.split('\n')
        .filter(line => line.includes('Guest:'))
        .join('. ');
      speak(guestText);
    }

    function speakFullDialogue() {
      if (!currentRoleplay) return;
      speak(currentRoleplay.dialogue.replace(/\n/g, '. '));
    }

    function markAsComplete() {
      alert("Progress saved! (We'll connect full tracking soon)");
      closeModal();
      loadRoleplays();
    }

    // Load on page open
    window.onload = loadRoleplays;
  </script>
